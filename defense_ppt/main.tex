\section{研究背景与意义}
% ═══════════════════════════════════════════════════════════════════════════

\begin{frame}{长期配置的核心挑战}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{为什么收益预测很难？}
      \begin{itemize}
        \item 均值–方差对收益率估计高度敏感
        \item 微小扰动导致权重剧烈变动（Michaud 1989）
        \item 收益率估计误差影响约为协方差的 \textbf{10 倍}
        \item 大量样本外检验：1/N 等权难以被击败
      \end{itemize}
      \vspace{0.5em}
      \textbf{实践层面的约束}
      \begin{itemize}
        \item 月度调仓 + 交易成本长期积累
        \item 相关结构时变、尾部依赖非线性
        \item 境内外 ETF 上市时间参差不齐
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{风险平价的核心思想}
        \centering
        以\textbf{风险贡献}替代资金金额\\[0.4em]
        \large
        $RC_i = w_i \cdot \dfrac{(\Sigma w)_i}{\sigma_p}$\\[0.6em]
        \normalsize
        目标：$RC_i = b_i \cdot \sigma_p$\\[0.4em]
        \small 对收益预测依赖低，可解释性强
      \end{block}
      \vspace{0.5em}
      \begin{alertblock}{全球 ETF 落地的问题}
        严格等式约束 + 权重上限 + CVaR 约束 + 换手惩罚\\
        → 无法纳入标准凸优化求解器
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{文献综述与理论基础}
% ═══════════════════════════════════════════════════════════════════════════

\begin{frame}{理论脉络：从均值–方差到风险预算}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{均值–方差框架的局限}
      \begin{itemize}
        \item Markowitz (1959)：二次规划奠基
        \item Michaud (1989)：估计误差放大效应
        \item DeMiguel 等 (2009)：14 种优化均未稳定击败 1/N
        \item Black-Litterman (1992)：需要先验，难以无模型验证
      \end{itemize}
      \vspace{0.6em}
      \textbf{协方差估计的研究进展}
      \begin{itemize}
        \item 随机矩阵理论（Laloux 等 1999）：大量特征值为噪声
        \item CM-IEWMA（Johansson 等 2023）：凸组合多个 EWMA
        \item 端到端深度学习（Bongiorno 等 2025）：难纳入硬约束
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{风险平价的理论发展}
      \begin{itemize}
        \item Qian (2005)：风险贡献平等 = 最优分散
        \item Maillard 等 (2010)：ERC 唯一性证明
        \item Roncalli (2013)：推广为完整风险预算框架
        \item Gambeta \& Kwon (2020)：\textbf{宽松风险平价}数学性质
      \end{itemize}
      \vspace{0.6em}
      \textbf{层次化方法}
      \begin{itemize}
        \item HRP（López de Prado 2016）：层次聚类，回避矩阵求逆
        \item HERC（Raffinot 2018）：簇间引入风险贡献逻辑
        \item \alert{局限}：难以直接纳入 CVaR 约束和换手惩罚
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{CVaR 与交易成本的理论基础}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{CVaR 的数学性质}\\[0.3em]
      Rockafellar \& Uryasev（2000；2002）将 CVaR 约束转化为\textbf{线性规划}：
      \[
        \mathrm{CVaR}_\alpha(w) = \eta + \frac{1}{(1-\alpha)T}\sum_{t=1}^T \max(-r_t^\top w - \eta,\, 0)
      \]
      \begin{itemize}
        \item 次可加性 + 凸性：天然适合约束优化
        \item 引入辅助变量 $\eta$ 后等价于线性不等式
        \item 与 $\ell_1$ 换手惩罚可统一纳入同一凸框架
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \textbf{交易成本建模}\\[0.3em]
      Almgren \& Chriss (2001)：换手速率与市场冲击成本的根本张力
      \vspace{0.5em}
      \begin{block}{本文采用线性近似}
        \centering
        $\mathrm{TC}_t = c \cdot \|w_t - w_{t-1}^+\|_1$\\[0.3em]
        \small $c = \txCostBps$ bps 单边\\
        月频调仓下合理性充分
      \end{block}
      \vspace{0.5em}
      \small 国内研究（华泰 2020--2024）：\\
      低利率环境下换手成本是\\
      风险平价落地的核心约束之一
    \end{column}
  \end{columns}
\end{frame}

\section{理论框架与模型方法}
% ═══════════════════════════════════════════════════════════════════════════

\begin{frame}{从标准风险平价到宽松风险平价}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{标准风险平价（ERC）}
      \[
        RC_i = w_i \cdot \frac{(\Sigma w)_i}{\sigma_p},\quad
        \sum_i RC_i = \sigma_p
      \]
      目标：$RC_i = b_i \sigma_p$（严格等式）\\[0.5em]
      \alert{问题}：加上权重上下限和组别约束后，直接求解不再可行
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{宽松风险平价（RRP）}\\[0.3em]
      将严格等式改为\textbf{偏离惩罚}：
      \[
        \min_w \sum_i \!\left(\frac{RC_i}{\sigma_p} - b_i\right)^{\!2}
        + \lambda_{\mathrm{reg}}\|w - w_0\|_2^2
      \]
      \[
        \textstyle\sum_i w_i = 1,\quad l_i \le w_i \le u_i
      \]
      \begin{block}{}
        允许在风险预算、权重稳定性\\
        和现实约束之间做折中
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Convex Adaptive Global RRP 目标函数}
  \small
  将风险预算、CVaR、换手成本和资产组约束统一入凸框架：
  \begin{align*}
    \min_w\; J(w) &= \lambda_{\mathrm{var}}\, w^\top\Sigma_t w
    + \lambda_{\mathrm{budget}}\|w - b_t\|_2^2 \\
    &\quad+ \bigl(\lambda_{\mathrm{turnover}} + \lambda_{\mathrm{tc}} c\bigr)\|w - w_{t-1}^+\|_1
    + \lambda_{\mathrm{cvar}}\,\mathrm{CVaR}_\alpha(w)
    - \lambda_{\mathrm{return}}\,\mu_t^\top w
  \end{align*}
  \[
    \textstyle\sum_i w_i = 1,\quad 0 \le w_i \le u_i,\quad L_g \le \textstyle\sum_{i\in g} w_i \le U_g
  \]
  \begin{center}
    \footnotesize
    \renewcommand{\arraystretch}{1.25}
    \begin{tabular}{l>{\raggedright\arraybackslash}p{4.8cm}>{\raggedright\arraybackslash}p{3.5cm}}
      \toprule
      模型 & 与 Global RRP 的差异 & 主要作用 \\
      \midrule
      Global RRP & 基准宽松风险预算 & 风险来源清晰，换手偏高 \\
      Convex Adaptive & 凸化近似 + $\ell_1$ 换手惩罚 + 资产组上限 & 大幅降低换手 \\
      \textbf{Improved Convex} & \textbf{+ CVaR 约束 + 冻结 OOS 参数} & \textbf{尾部控制 + 防过拟合} \\
      HERC & 层次聚类，回避矩阵求逆 & 基准对比 \\
      \bottomrule
    \end{tabular}
  \end{center}
\end{frame}

\begin{frame}{参数冻结：防止过拟合}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{冻结样本外（Frozen OOS）流程}
      \begin{enumerate}
        \item 在训练集内网格搜索超参数候选
        \item 筛选出 OOS 期表现稳健的候选集
        \item \textbf{冻结参数}后在完整评价期重新运行
        \item 所有报告数字均来自冻结参数下的结果
      \end{enumerate}
      \vspace{0.4em}
      \begin{block}{冻结参数（candidate\_23）}
        \small
        \begin{tabular}{ll}
          历史窗口 & 252 个交易日 \\
          协方差估计 & EWMA \\
          单资产最大权重 & 45\% \\
          换手上限 & 80\% \\
          换手惩罚 $\lambda_{\mathrm{to}}$ & 0.020 \\
          CVaR 惩罚 $\lambda_{\mathrm{cvar}}$ & 0.080 \\
          预算惩罚 $\lambda_{\mathrm{budget}}$ & 0.100 \\
          CVaR 置信水平 $\alpha$ & 0.95 \\
          收益奖励 $\lambda_{\mathrm{return}}$ & 0.060 \\
        \end{tabular}
      \end{block}
    \end{column}
    \begin{column}{0.42\textwidth}
      \textbf{为什么冻结参数重要？}
      \begin{itemize}
        \item 超参数搜索本身会对样本过拟合
        \item 如果参数随时间滚动优化 → 前视偏差
        \item 冻结后：模型在测试集上的表现是\textbf{真正的样本外结果}
      \end{itemize}
      \vspace{0.6em}
      \begin{alertblock}{诚实的结论边界}
        所有数字均条件于\\
        \textbf{2019--2026 年这一特定样本区间}\\
        和上述约束参数设定，\\
        不代表对未来绩效的预测
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{关键发现：三维权衡的改善}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{换手率的问题（Global RRP）}
      \begin{itemize}
        \item 月均换手 \globalMonthlyTurnover → 年化换手 \textasciitilde267\%
        \item 成本拖累明显：净收益仅 \globalNetReturn
        \item 严格风险贡献约束 → 每月大幅再平衡
      \end{itemize}
      \vspace{0.5em}
      \textbf{Improved Convex 的改进}
      \begin{itemize}
        \item 换手压至 \improvedMonthlyTurnover（降低约 \textbf{10 倍}）
        \item Sharpe 从 \globalSharpe{} 升至 \textbf{\improvedSharpe}
        \item Sortino 从 \globalSortino{} 升至 \textbf{\improvedSortino}
        \item 最大回撤从 \globalMaxDD{} 收窄至 \textbf{\improvedMaxDD}
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{HERC 的 Sharpe 从何而来？}
      \begin{itemize}
        \item 表观 Sharpe = \hercSharpe，看似不错
        \item 但年化波动率仅 \hercVolatility
        \item 组合近乎全仓债券/货币市场
        \item 这是\alert{极低波动的人工压制}，\\
              \textbf{不是}多资产分散的结果
        \item 净年化收益仅 \hercNetReturn，\\
              远低于 60/40 的 \sixtyFortyNetReturn
      \end{itemize}
      \vspace{0.3em}
      \begin{block}{核心结论}
        Improved Convex 在\textbf{收益、回撤、换手}三维度均无明显短板，HERC 的高 Sharpe 不具可比性
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{多维稳健性检验框架}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{1. Walk-forward 验证}\\[0.3em]
      滚动子区间（多个训练-测试窗口）：
      \begin{itemize}
        \item Sharpe 均值：\walkforwardSharpeMean{}，中位数：\walkforwardSharpeMedian
        \item 净收益均值：\walkforwardNetreturnMean{}，中位数：\walkforwardNetreturnMedian
        \item 最大回撤均值：\walkforwardMaxddMean{}，中位数：\walkforwardMaxddMedian
        \item 结论：中位数 Sharpe 约 1.0，跨区间相对稳定
      \end{itemize}
      \vspace{0.4em}
      \textbf{2. 参数扰动分析}
      \begin{itemize}
        \item 对 5 个超参数分别 ±20\% 扰动
        \item Sharpe 和换手率的分布范围
        \item 识别对哪些参数最敏感
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{3. 协方差估计器对比}
      \begin{itemize}
        \item 样本协方差 vs. EWMA vs. 收缩估计
        \item 本文基准：EWMA（冻结参数选定）
        \item 其他估计器下结论方向一致
      \end{itemize}
      \vspace{0.4em}
      \textbf{4. 成本分析}
      \begin{itemize}
        \item 交易成本盈亏平衡分析
        \item Improved vs. Global RRP：需超过 \textbf{50 bps} 单边成本方可被击败
        \item 月频 3 bps 远低于此阈值
      \end{itemize}
      \vspace{0.4em}
      \textbf{5. 过拟合诊断}
      \begin{itemize}
        \item OOS 候选参数验证
        \item 确认结论非样本内过拟合产物
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{结论的适用边界}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{结论成立的条件}
      \begin{itemize}
        \item 样本区间：\evalStartDate{} -- \evalEndDate
        \item 资产池：\etfCount{} 支可交易 A 股/海外 ETF
        \item 月度调仓，3 bps 单边成本
        \item 冻结参数 candidate\_23
      \end{itemize}
      \vspace{0.5em}
      \textbf{哪些方向可能改变结论？}
      \begin{itemize}
        \item \alert{市场机制切换}：熊市/危机下 CVaR 约束有效性存疑
        \item \alert{资产池扩充}：加入更多品种可能改变聚类结构
        \item \alert{高成本环境}：若实际成本远超 3 bps，换手节约的价值更大
        \item \alert{不同频率}：日频调仓下换手惩罚参数需重新校准
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{研究局限（三项）}
        \begin{enumerate}
          \item 数据期间较短（89 个月），包含低利率→加息周期，与历史长期均值偏离
          \item 协方差矩阵估计仍依赖历史数据；若相关结构快速变化，EWMA 存在滞后
          \item 线性成本近似在大规模执行场景下需要精细化（市场冲击、流动性约束）
        \end{enumerate}
      \end{alertblock}
      \vspace{0.4em}
      \begin{block}{未来研究方向}
        \small
        \begin{itemize}
          \item 因子层 CVaR 约束（资产 + 因子双层预算）
          \item 机器学习协方差估计器与凸优化结合
          \item 更长历史区间（含 2015 股灾、2020 新冠）
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{三条主要结论}
  \begin{swufekeypoints}
    \swufepoint{CVaR + 换手惩罚有效改善三维权衡}{01}{Improved Convex 将换手率从 \globalMonthlyTurnover{} 降至 \improvedMonthlyTurnover，净 Sharpe 升至 \improvedSharpe，最大回撤收窄至 \improvedMaxDD。Sharpe 提升在统计上显著（$p=\improvedVsGlobalPValue$）}
