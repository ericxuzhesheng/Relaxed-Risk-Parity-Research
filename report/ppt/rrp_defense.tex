\section{凸优化方法}
\begin{frame}{风险预算参考}
\[
\min_{x>0}\;\frac12x^\mathsf{T}\Sigma_t x-\sum_i b_i\log x_i,
\qquad q_t=\frac{x_t}{\mathbf1^\mathsf{T}x_t}.
\]
\begin{itemize}
\item 对全部当期合格资产设等风险预算
\item 最多 \lookbackDays{} 个历史观察，采用 Ledoit--Wolf 收缩协方差
\item 收益估计采用 20 个交易日半衰期
\item 所有估计只使用调仓日前历史数据
\item 正半定二次项与负对数项构成凸目标
\end{itemize}
\end{frame}
\begin{frame}{Global RRP 的凸松弛}
\[
\min_w\;\|w-q_t\|_2^2+\lambda_{v,t}\frac{w^\mathsf{T}\Sigma_t w}{v_t}
+\lambda_{r,t}\left[\max\left(0,\frac{R_t-\mu_t^\mathsf{T}w}{s_t}\right)\right]^2
\]
\[
\mathbf1^\mathsf{T}w=1,\quad w\geq0,\quad v_t=e_t^\mathsf{T}\Sigma_t e_t.
\]
\begin{itemize}
\item $e_t$ 为当期合格资产的等权组合
\item $R_t=\mu_t^\mathsf{T}q_t$，取当期可行参考组合的预测收益
\item $\lambda_{v,t}$ 与 $\lambda_{r,t}$ 按年度更新并在年内冻结
\item 多头、无杠杆；不再叠加求解后的风险缩放
\end{itemize}
\end{frame}

\begin{frame}{为什么采用无杠杆配置}
\begin{itemize}
\item 境内ETF机会集合：权益工具较多，独立利率、通胀与货币对冲风险来源较有限
\item 多头单位预算将风险规模与资本占用相连，低波动配置需要更多资本
\item 桥水全球机构框架使用更广泛的债券、衍生品及融资安排调节风险规模
\item 本文完成给定资本预算下的配置；融资扩展需同步计入利息、保证金与展期
\end{itemize}
\medskip\scriptsize 来源：Bridgewater（2012）；Frazzini与Pedersen（JFE，2014）；Brunnermeier与Pedersen（RFS，2009）。
\end{frame}
\begin{frame}{Ledoit--Wolf：收缩什么，为什么收缩}
\[
\widetilde\Sigma=(1-\delta)S^{\rm ML}+\delta\tau I,\qquad
\tau=\operatorname{tr}(S^{\rm ML})/n
\]
\begin{itemize}
\item 有限样本下平衡估计偏差与噪声，改善矩阵稳定性
\item 方差向各资产平均方差靠拢，协方差向零靠拢
\item 强度$\delta$由当期历史数据估计，与年度优化惩罚系数分开
\item 货币ETF原始波动很低，等方差目标会提高其估计波动
\end{itemize}
\scriptsize 经验矩阵使用中心化日收益和$M$分母；年化后再作数值修复。来源：Ledoit与Wolf（Journal of Multivariate Analysis，2004）。
\end{frame}
\begin{frame}{收缩估计怎样影响风险判断}
\resizebox{\textwidth}{!}{\input{../thesis_latex/generated_covariance_interpretation.tex}}
\par\medskip
\[
w^\mathsf T\widetilde\Sigma w=(1-\delta)w^\mathsf T S^{\rm ML}w+\delta\tau\|w\|_2^2
\]
\begin{itemize}
\item 共同方差目标使资本集中度进入估计风险，现金集中组合也受影响
\item 表中是调仓时估计值，经济效果由实际收益、波动与回撤评价
\end{itemize}
\end{frame}

\begin{frame}{Brinson归因：解释相对基准的收益差}
\small
\begin{align*}
A_g&=(w^P_g-w^B_g)(r^B_g-R^B) &&\text{配置}\\
S_g&=w^B_g(r^P_g-r^B_g) &&\text{组内选择}\\
I_g&=(w^P_g-w^B_g)(r^P_g-r^B_g) &&\text{交互}
\end{align*}
\begin{itemize}
\item 三项之和解释毛收益差，加入费用差后逐日核对净收益差
\item 本文的选择效应指组内ETF配置；风险贡献另由权重和协方差计算
\item 现金超配的贡献取决于现金收益相对基准整体收益的高低
\end{itemize}
\scriptsize 来源：Brinson与Fachler（JPM，1985）；Brinson等（FAJ，1986）；Bacon（CFA Institute，2019）。
\end{frame}

\resizebox{\textwidth}{!}{\input{../thesis_latex/generated_brinson_summary.tex}}

\item 表中为平均单日基点；下跌日按沪深300ETF当日负收益事后划分
\item 下跌日货币配置项：相对原始等权为\attrEqualCashDown{}，相对风险匹配等权为\attrMatchedCashDown{}
\item 风险匹配基准本身持有较多现金，评价应同时观察国债、信用债和组内配置
\end{itemize}
\medskip\scriptsize
共\attrDays{}个交易日，其中\attrDownDays{}个下跌日。单日算术归因与年化、复合收益分别报告。
\end{frame}
