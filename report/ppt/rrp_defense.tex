\section{凸优化方法}
\begin{frame}{风险预算参考}
\[
\min_{x>0}\;\frac12x^\mathsf{T}\Sigma_t x-\sum_i b_i\log x_i,
\qquad q_t=\frac{x_t}{\mathbf1^\mathsf{T}x_t}.
\]
\begin{itemize}
\item 对全部当期合格资产设等风险预算
\item 最多 \lookbackDays{} 个历史观察，采用 Ledoit--Wolf 收缩协方差
\item 收益估计采用 20 个交易日半衰期
\item 所有估计只使用调仓日前历史数据
\item 正半定二次项与负对数项构成凸目标
\end{itemize}
\end{frame}
\begin{frame}{Global RRP 的凸松弛}
\[
\min_w\;\|w-q_t\|_2^2+\lambda_{v,t}\frac{w^\mathsf{T}\Sigma_t w}{v_t}
+\lambda_{r,t}\left[\max\left(0,\frac{R_t-\mu_t^\mathsf{T}w}{s_t}\right)\right]^2
\]
\[
\mathbf1^\mathsf{T}w=1,\quad w\geq0,\quad v_t=e_t^\mathsf{T}\Sigma_t e_t.
\]
\begin{itemize}
\item $e_t$ 为当期合格资产的等权组合
\item $R_t=\mu_t^\mathsf{T}q_t$，取当期可行参考组合的预测收益
\item $\lambda_{v,t}$ 与 $\lambda_{r,t}$ 按年度更新并在年内冻结
\item 多头、无杠杆；不再叠加求解后的风险缩放
\end{itemize}
\end{frame}
\begin{frame}{年度参数校准}
\begin{columns}[T]

\end{frame}

\section{验证与边界}
\begin{frame}{分自然年结果}
\centering
{\scriptsize\renewcommand{\arraystretch}{0.88}\input{../thesis_latex/generated_primary_annual.tex}}
\par\medskip
\small 最后一年仅截至 \evalEndDate，表中为年化指标。
\end{frame}
\begin{frame}{实现检查与资产参与}
\begin{itemize}
\item 每期检查 DCP 凸性、求解状态、预算及非负约束
\item 信息截止时间严格早于调仓日
\item 年度验证期末严格早于参数生效日
\item 使用交易前漂移权重计算换手与成本
\item $5\times10^{-5}$ 只用于识别数值残差，不是最低持仓约束
\end{itemize}
\end{frame}
\begin{frame}{研究边界}
\begin{itemize}
\item 规格选择利用历史结果，尚需新增数据检验
\item 均值和协方差估计存在误差
\item 固定成本未证明实际成交、容量与市场冲击
\item 未处理负债、资本占用及久期缺口
\item 资产曾被使用，不等于风险贡献充分分散
\item 年度候选缺乏显著区分，参数不能解释为唯一最优值
\end{itemize}
\end{frame}
\section{结论}
\begin{frame}{研究结论}
\begin{itemize}
\item Global RRP 采用凸优化生成周频、多头、无杠杆组合
\item 净年化收益 \globalNetReturn，夏普 \globalSharpe
\item 年化波动 \globalVolatility，最大回撤 \globalMaxDD
\item 年度参数只使用生效日前数据，但统计识别仍然较弱
\item \primaryAssetsUsed{} 只 ETF 曾形成超过数值容差的持仓
\end{itemize}
\end{frame}
