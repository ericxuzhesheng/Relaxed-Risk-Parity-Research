\section{研究背景与意义}
% ════════════════════════════════════════════════════════════════════════════

\begin{frame}{长期配置的核心挑战}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{为什么不能直接用均值—方差？}
      \begin{itemize}
        \item Michaud（1989）：收益率估计误差远高于协方差估计误差
        \item 微小预测扰动可能导致权重剧烈漂移
        \item DeMiguel 等（2009）：多类优化策略未稳定击败 1/N
        \item 机构配置更需要\textbf{风险可归因}的框架
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{全球 ETF 落地的三重难题}
      \begin{itemize}
        \item 相关结构\textbf{时变}：固定窗口协方差估计滞后
        \item 严格 ERC + 组别上限 + CVaR：形成\textbf{非凸}约束
        \item 月频调仓叠加 \txCostBps{} bps 成本：净收益被持续侵蚀
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{block}{本页要点}
    \centering
    长期资产配置的难点不是单期收益排序，而是如何在不确定环境下保持风险来源清晰、约束可执行、换手可控制。
  \end{block}
\end{frame}

\begin{frame}{风险平价的核心逻辑与本文解法}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{风险平价的核心逻辑}
        \centering\large
        $RC_i = w_i \cdot \dfrac{(\Sigma w)_i}{\sigma_p}$\\[0.5em]
        \normalsize 目标：$RC_i = b_i \cdot \sigma_p$\\[0.3em]
        \small 以\textbf{风险贡献}替代资金金额分配\\
        对收益预测依赖低，逻辑清晰
      \end{block}
      \vspace{0.4em}
      \begin{alertblock}{本文的解法}
        宽松 RRP + CVaR 尾部约束 +\\
        $\ell_1$ 换手惩罚 →\\
        \textbf{统一可稳定求解的凸优化框架}
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{方法含义}
      \begin{itemize}
        \item 用风险预算偏离惩罚替代严格等式
        \item 将 CVaR 尾部风险控制纳入约束
        \item 用 $\ell_1$ 换手惩罚提高可实施性
        \item 最终权重由透明优化问题生成
      \end{itemize}
      \vspace{0.5em}
      \begin{block}{研究定位}
        \small
        本文关注长期机构配置中的可解释、可约束、可执行风险预算框架，不把模型解释为短期择时交易策略。
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{文献综述与理论基础}
% ════════════════════════════════════════════════════════════════════════════

\begin{frame}{均值—方差与估计误差：理论起点}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{均值—方差与估计误差}
      \begin{itemize}
        \item Markowitz（1959）：均值—方差模型奠定现代组合选择基础
        \item Jorion（1986）：Bayes-Stein 收缩量化参数不确定性
        \item 核心问题：收益预测误差会被优化器放大
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{协方差估计进展}
      \begin{itemize}
        \item 随机矩阵理论（Laloux 1999）：大量特征值为噪声
        \item CM-IEWMA（Johansson 2023）：凸组合多个 EWMA 估计器
        \item 端到端深度学习（Bongiorno 2025）：预测能力提升，但硬约束接口仍需处理
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{alertblock}{共同结论}
    收益预测不确定性高，单纯追求均值—方差最优容易产生不稳定权重，因此本文转向风险预算与约束优化框架。
  \end{alertblock}
\end{frame}

\begin{frame}[shrink=5]{风险预算与层次化方法：本文连接点}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{风险平价与宽松风险预算}
      \begin{itemize}
        \item Qian（2005）：风险贡献平等 = 最优分散
        \item Maillard 等（2010）：ERC 唯一性证明
        \item Gambeta \& Kwon（2020）：\textbf{宽松 RRP} 数学性质
        \item Ielpo 等（2024）：\textbf{Regime Parity} 状态平价，缓解无条件矩低估危机协动
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{层次化方法}
      \begin{itemize}
        \item HRP（López de Prado 2016）：层次聚类，回避矩阵求逆
        \item HERC（Raffinot 2018）：簇间引入风险贡献逻辑
        \item 在本文中作为层次化风险分配基准
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.2em}
  \begin{block}{本文连接点}
    \centering
    风险预算提供可解释目标，凸优化框架提供 CVaR、换手率和资产组约束的统一实现方式。
  \end{block}
\end{frame}

\begin{frame}[shrink=2]{CVaR 约束与交易成本的理论依据}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{CVaR 的凸性与线性重表述}\\[0.3em]
      Rockafellar \& Uryasev（2000；2002）：
      \begin{center}
      \resizebox{0.98\linewidth}{!}{$
      \begin{aligned}
        \mathrm{CVaR}_\alpha(w)
        &= \min_\eta \Bigl[
        \eta + \tfrac{1}{(1-\alpha)T}
        \sum_t \max(-r_t^\top w-\eta,0)
        \Bigr]
      \end{aligned}
      $}
      \end{center}
      \begin{itemize}
        \item 次可加性 + 凸性：\textbf{天然适合约束优化}
        \item 引入辅助变量 $\eta$ 后等价为\textbf{线性不等式组}
        \item 与 $\ell_1$ 换手惩罚可统一在同一 QP/SOCP 框架
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{交易成本建模}\\[0.3em]
      Almgren \& Chriss（2001）：换手速率与市场冲击的根本张力
      \vspace{0.4em}
      \begin{block}{本文：线性成本近似}
        \centering
        $\mathrm{TC}_t = c \cdot \|w_t - w_{t-1}^+\|_1$\\[0.3em]
        \small $c = \txCostBps$ bps 单边\\月频调仓下误差可控
      \end{block}
      \vspace{0.4em}
      \small \textbf{国内研究（华泰 2020—2024）指出}：低利率环境下换手成本是风险平价落地的核心约束，月频再平衡的成本积累不可忽视
    \end{column}
  \end{columns}
\end{frame}

\section{理论框架与模型方法}
% ════════════════════════════════════════════════════════════════════════════

\begin{frame}[shrink=3]{从严格 ERC 到宽松风险平价}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \textbf{标准风险平价（ERC）}\\[0.2em]
      风险贡献：$RC_i = w_i \cdot \dfrac{(\Sigma w)_i}{\sigma_p}$\\[0.4em]
      严格等式约束：$RC_i = b_i \sigma_p$\\[0.4em]
      \alert{困难}：同时加权重上下限、资产组约束后，等式约束使问题\textbf{非凸}，标准求解器无法直接处理

      \vspace{0.6em}
      \textbf{宽松 RRP：严格等式 → 偏离惩罚}
      \[
        \min_w \sum_i \!\Bigl(\tfrac{RC_i}{\sigma_p} - b_i\Bigr)^{\!2}
        + \lambda_{\text{reg}}\|w - w_0\|_2^2
      \]
      允许可量化的预算偏离，换取约束灵活性
    \end{column}
    \begin{column}{0.51\textwidth}
      \textbf{Convex Adaptive Global RRP}\\[0.3em]
      将风险预算思想凸化，统一纳入多个约束：
      \scriptsize
      \begin{align*}
        \min_w\; J(w)
          &= \underbrace{\lambda_v w^\top\Sigma_t w}_{\text{波动率项}}
           + \underbrace{\lambda_b\|w - b_t\|_2^2}_{\text{预算偏离}} \\[2pt]
          &\;+ \underbrace{\lambda_{to}\|w - w_{t-1}^+\|_1}_{\text{换手惩罚}}
           + \underbrace{\lambda_c\,\mathrm{CVaR}_\alpha(w)}_{\text{尾部约束}}
           - \underbrace{\lambda_r\,\mu_t^\top w}_{\text{收益奖励}}
      \end{align*}
      \small
      \vspace{0.1em}
      约束集：$\sum_i w_i=1,\ 0\le w_i\le u_i,\ L_g\le\sum_{i\in g}w_i\le U_g$
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{三条核心结论}
  \begin{swufekeypoints}
    \swufepoint{三维权衡形成折中}{01}{Improved Convex 月换手从 \globalMonthlyTurnover{} 降至 \improvedMonthlyTurnover{}，净 Sharpe 升至 \improvedSharpe{}（统计显著，$p=\improvedVsGlobalPValue$），最大回撤收窄至 \improvedMaxDD{}——优势在于低波动、浅回撤与可控换手，代价是权益强势阶段收益弹性有限}
    \swufepoint{HERC 需结合权重解释}{02}{HERC 年化波动率仅 \hercVolatility{}，净收益 \hercNetReturn{}，其 Sharpe = \hercSharpe{} 主要来自接近全仓债券的低波动结构；因此应作为层次化配置基准，而非与主模型作单一指标排序}
    \swufepoint{结论有明确的适用边界}{03}{所有数字均条件于 \evalStartDate{}—\evalEndDate{} 这一特定样本区间和滚动选择规则；Walk-forward、参数扰动和调仓频率检验用于约束结论强度。月度调仓本次恰好排名第一，但实施规则并非按事后收益选择，也不代表对未来绩效的预测}
  \end{swufekeypoints}
\end{frame}

\begin{frame}{研究局限与未来方向}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{三项研究局限}

        \item \textbf{样本期偏短}：\evalMonths{} 个月，包含低利率→快速加息完整周期，与更长历史均值存在结构性偏差
        \item \textbf{协方差估计滞后}：EWMA 在市场机制快速切换时存在系统性滞后；更先进的估计器尚未纳入
        \item \textbf{成本模型简化}：线性近似在大规模执行场景下忽视了市场冲击和流动性约束
      \end{enumerate}
