\section{研究背景与意义}
% ════════════════════════════════════════════════════════════════════════════

\begin{frame}{长期配置的核心挑战}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{为什么不能直接用均值—方差？}
      \begin{itemize}
        \item Michaud（1989）：收益率估计误差远高于协方差估计误差
        \item 微小预测扰动可能导致权重剧烈漂移
        \item DeMiguel 等（2009）：多类优化策略未稳定击败 1/N
        \item 机构配置更需要\textbf{风险可归因}的框架
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{全球 ETF 落地的三重难题}
      \begin{itemize}
        \item 相关结构\textbf{时变}：固定窗口协方差估计滞后
        \item 严格 ERC + 组别上限 + CVaR：形成\textbf{非凸}约束
        \item 月频调仓叠加 \txCostBps{} bps 成本：净收益被持续侵蚀
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{block}{本页要点}
    \centering
    长期资产配置的难点不是单期收益排序，而是如何在不确定环境下保持风险来源清晰、约束可执行、换手可控制。
  \end{block}
\end{frame}

\begin{frame}{风险平价的核心逻辑与本文解法}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{风险平价的核心逻辑}
        \centering\large
        $RC_i = w_i \cdot \dfrac{(\Sigma w)_i}{\sigma_p}$\\[0.5em]
        \normalsize 目标：$RC_i = b_i \cdot \sigma_p$\\[0.3em]
        \small 以\textbf{风险贡献}替代资金金额分配\\
        对收益预测依赖低，逻辑清晰
      \end{block}
      \vspace{0.4em}
      \begin{alertblock}{本文的解法}
        宽松 RRP + CVaR 尾部约束 +\\
        $\ell_1$ 换手惩罚 →\\
        \textbf{统一可稳定求解的凸优化框架}
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{方法含义}
      \begin{itemize}
        \item 用风险预算偏离惩罚替代严格等式
        \item 将 CVaR 尾部风险控制纳入约束
        \item 用 $\ell_1$ 换手惩罚提高可实施性
        \item 最终权重由透明优化问题生成
      \end{itemize}
      \vspace{0.5em}
      \begin{block}{研究定位}
        \small
        本文关注长期机构配置中的可解释、可约束、可执行风险预算框架，不把模型解释为短期择时交易策略。
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{研究问题与贡献}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \textbf{研究问题}

        \item 在凸优化中同时处理风险预算、CVaR、换手和组别边界
        \item 用滚动样本外路径与多类诊断约束结论
      \end{itemize}

      \begin{block}{评价窗口}

\section{文献综述与理论基础}
% ════════════════════════════════════════════════════════════════════════════

\begin{frame}{文献脉络与本文切入点}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{从均值—方差到风险预算}
      \begin{itemize}
        \item Markowitz（1959）建立收益—风险优化框架
        \item Michaud（1989）指出收益估计误差会放大权重波动
        \item Qian（2005）与 Maillard 等（2010）改用风险贡献组织组合
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{约束优化与层次化基准}
      \begin{itemize}
        \item Gambeta 与 Kwon（2020）用惩罚项放松严格风险贡献等式
        \item HRP/HERC 回避矩阵求逆，但难直接加入 CVaR 与换手约束
        \item 本文保留透明的凸优化权重生成过程
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{alertblock}{切入点}
    风险预算提供可解释目标，凸优化同时容纳 CVaR、换手率和资产组边界。
  \end{alertblock}
\end{frame}

\begin{frame}[shrink=2]{CVaR 约束与交易成本的理论依据}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{CVaR 的凸性与线性重表述}\\[0.3em]
      Rockafellar \& Uryasev（2000；2002）：
      \begin{center}
      \resizebox{0.98\linewidth}{!}{$
      \begin{aligned}
        \mathrm{CVaR}_\alpha(w)
        &= \min_\eta \Bigl[
        \eta + \tfrac{1}{(1-\alpha)T}
        \sum_t \max(-r_t^\top w-\eta,0)
        \Bigr]
      \end{aligned}
      $}
      \end{center}
      \begin{itemize}
        \item 次可加性 + 凸性：\textbf{天然适合约束优化}
        \item 引入辅助变量 $\eta$ 后等价为\textbf{线性不等式组}
        \item 与 $\ell_1$ 换手惩罚可统一在同一 QP/SOCP 框架
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{交易成本建模}\\[0.3em]
      Almgren \& Chriss（2001）：换手速率与市场冲击的根本张力
      \vspace{0.4em}
      \begin{block}{本文：线性成本近似}
        \centering
        $\mathrm{TC}_t = c \cdot \|w_t - w_{t-1}^+\|_1$\\[0.3em]
        \small $c = \txCostBps$ bps 单边\\月频调仓下误差可控
      \end{block}
      \vspace{0.4em}
      \small \textbf{国内研究（华泰 2020—2024）指出}：低利率环境下换手成本是风险平价落地的核心约束，月频再平衡的成本积累不可忽视
    \end{column}
  \end{columns}
\end{frame}

\section{理论框架与模型方法}
% ════════════════════════════════════════════════════════════════════════════

\begin{frame}[shrink=3]{从严格 ERC 到宽松风险平价}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \textbf{标准风险平价（ERC）}\\[0.2em]
      风险贡献：$RC_i = w_i \cdot \dfrac{(\Sigma w)_i}{\sigma_p}$\\[0.4em]
      严格等式约束：$RC_i = b_i \sigma_p$\\[0.4em]
      \alert{困难}：同时加权重上下限、资产组约束后，等式约束使问题\textbf{非凸}，标准求解器无法直接处理

      \vspace{0.6em}
      \textbf{宽松 RRP：严格等式 → 偏离惩罚}
      \[
        \min_w \sum_i \!\Bigl(\tfrac{RC_i}{\sigma_p} - b_i\Bigr)^{\!2}
        + \lambda_{\text{reg}}\|w - w_0\|_2^2
      \]
      允许可量化的预算偏离，换取约束灵活性
    \end{column}
    \begin{column}{0.51\textwidth}
      \textbf{Convex Adaptive Global RRP}\\[0.3em]
      将风险预算思想凸化，统一纳入多个约束：
      \scriptsize
      \begin{align*}
        \min_w\; J(w)
          &= \underbrace{\lambda_v w^\top\Sigma_t w}_{\text{波动率项}}
           + \underbrace{\lambda_b\|w - b_t\|_2^2}_{\text{预算偏离}} \\[2pt]
          &\;+ \underbrace{\lambda_{to}\|w - w_{t-1}^+\|_1}_{\text{换手惩罚}}
           + \underbrace{\lambda_c\,\mathrm{CVaR}_\alpha(w)}_{\text{尾部约束}}
           - \underbrace{\lambda_r\,\mu_t^\top w}_{\text{收益奖励}}
      \end{align*}
      \small
      \vspace{0.1em}
      约束集：$\sum_i w_i=1,\ 0\le w_i\le u_i,\ L_g\le\sum_{i\in g}w_i\le U_g$
    \end{column}
  \end{columns}
\end{frame}

\section{稳健性检验}
% ════════════════════════════════════════════════════════════════════════════

\begin{frame}{稳健性检验体系：六个维度}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{enumerate}
        \item \textbf{Walk-forward 子区间}：滚动训练-测试窗口，检验跨期稳定性
        \item \textbf{参数扰动}：对 5 个超参数各自 ±20\% 扰动，观察 Sharpe 和换手的分布范围
        \item \textbf{交易成本敏感性}：从 0 到 50+ bps 扫描，确定盈亏平衡阈值
      \end{enumerate}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{enumerate}\setcounter{enumi}{3}
        \item \textbf{调仓频率敏感性}：周度、双周、月度、季度仅改变调仓日历
        \item \textbf{Bootstrap 分布}：对月度收益序列重抽样，获取 Sharpe 和最大回撤的经验分布
        \item \textbf{协方差估计器}：比较样本协方差、EWMA（基准）、收缩估计三种方案
      \end{enumerate}
      \vspace{0.4em}
      \begin{alertblock}{目标}
        \small 将结论\textbf{明确限定}在特定样本区间和参数设定，避免过度外推
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{主要结论}
% ════════════════════════════════════════════════════════════════════════════

\begin{frame}{三条核心结论}
  \begin{swufekeypoints}
    \swufepoint{可实施性}{01}{Improved Convex 的 Sharpe 为 \improvedSharpe{}，最大回撤 \improvedMaxDD{}，月均换手 \improvedMonthlyTurnover{}。三项指标共同改善，夏普差异检验显著。}
    \swufepoint{基准解释}{02}{HERC 的 \hercSharpe{} Sharpe 来自债券集中和 \hercVolatility{} 的极低波动。权重结构比单一排名更能说明模型差异。}
    \swufepoint{适用边界}{03}{结论条件于 \evalStartDate{}—\evalEndDate{} 的历史样本、无杠杆约束和 3 bps 成本。权益强势期可能明显滞后。}
  \end{swufekeypoints}
\end{frame}

\begin{frame}{研究边界与后续方向}
  \centering
  \small
  \renewcommand{\arraystretch}{1.45}
  \begin{tabular}{p{0.16\textwidth}p{0.35\textwidth}p{0.35\textwidth}}
    \toprule
    \textbf{维度} & \textbf{当前边界} & \textbf{下一步} \\
    \midrule
    执行成本 & 固定 3 bps，未计容量与冲击 & 引入价差、成交量和非线性冲击 \\
    机构约束 & 未建模负债与资本占用 & 加入现金流、久期缺口和监管约束 \\
    样本结构 & 部分 ETF 历史短，相关性会变化 & 延长样本并检验因子层风险预算 \\
    资产覆盖 & 尚未纳入公募 REITs ETF & 流动性和历史充分后纳入资产池评审 \\
    \bottomrule
  \end{tabular}
  \vspace{1em}
  \begin{alertblock}{结论边界}
    \centering 历史证据支持可实施性，不构成未来收益保证。
  \end{alertblock}
\end{frame}
